Las descripciones de cada atributo son: - id: identificador unico del
paciente

\begin{itemize}
\item
  age\_days: edad en días - age\_year: edad en años
\item
  gender: genero (1 = mujer, 2 = hombre)
\item
  height: estatura en cm - weight: pero en kg
\item
  ap\_hi: presion arterial sistolica
\item
  ap\_lo: presion arterial diastolica
\item
  cholesterol: colesterol (1 = normal, 2 = arriba normal, 3 = muy alto)
\item
  gluc: nivel de glucosa (1 = normal, 2 = arriba normal, 3 = muy alto)
\item
  smoke: si el paciente fuma o no (0 = no, 1 = sí))
\item
  alco: si el paciente toma alcohol o no (0 = no, 1 = sí)
\item
  active: caracteristica binaria de la actividad física (0 = vida
  pasiva, 1 = vida activa)
\item
  cardio: Variable de interes si el paciente cuenta con enfermedad
  cardiovascular o no (0 = no, 1 = yes)
\end{itemize}
